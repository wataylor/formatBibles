Sub RemoveCurlyBracesContent()
    With Selection.Find
        .ClearFormatting
        .Replacement.ClearFormatting
        .Text = "\{*\}"
        .Replacement.Text = ""
        .Forward = True
        .Wrap = wdFindContinue
        .Format = False
        .MatchWildcards = True
    End With
    Selection.Find.Execute Replace:=wdReplaceAll
End Sub
